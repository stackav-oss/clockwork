template <size_t size>
class AlignmentTest_FixedSoaElementRef<::test::AlignmentTest, size>
{

  // Field accessors

  [[nodiscard]] const int32_t& get_four_byte() const;

  // Assignment operators
  AlignmentTest_FixedSoaElementRef& operator=(const ::clockwork::Tap<::clockwork::Tachyon<::test::AlignmentTest>>& tap);
  AlignmentTest_FixedSoaElementRef& operator=(const AlignmentTest_FixedSoaElementRef& other);
  AlignmentTest_FixedSoaElementRef& operator=(AlignmentTest_FixedSoaElementRef&& other);
  AlignmentTest_FixedSoaElementRef& operator=(const ::clockwork::TapInit<::clockwork::Tachyon<::test::AlignmentTest>>& init);

  // Conversion to Tap<Tachyon<SoaTestSchema>>
  explicit operator ::clockwork::Tap<::clockwork::Tachyon<::test::AlignmentTest>>() const;

  bool operator==(const AlignmentTest_FixedSoaElementRef& other) const;

  // Swap for lvalue references (called by std::swap)
  friend void swap(AlignmentTest_FixedSoaElementRef& ref_a, AlignmentTest_FixedSoaElementRef& ref_b) noexcept
  {
    const auto temp_two_byte = ref_a.get_two_byte();
    const auto temp_eight_byte = ref_a.get_eight_byte();
    const auto temp_one_byte = ref_a.get_one_byte();
    const auto temp_four_byte = ref_a.get_four_byte();
    ref_a.get_mutable_two_byte() = ref_b.get_two_byte();
    ref_a.get_mutable_eight_byte() = ref_b.get_eight_byte();
    ref_a.get_mutable_one_byte() = ref_b.get_one_byte();
    ref_a.get_mutable_four_byte() = ref_b.get_four_byte();
    ref_b.get_mutable_two_byte() = temp_two_byte;
    ref_b.get_mutable_eight_byte() = temp_eight_byte;
    ref_b.get_mutable_one_byte() = temp_one_byte;
    ref_b.get_mutable_four_byte() = temp_four_byte;
  }

  // Swap for rvalue references (called by std::iter_swap via *it)
  friend void swap(
    AlignmentTest_FixedSoaElementRef&& ref_a,           // NOLINT(cppcoreguidelines-rvalue-reference-param-not-moved)
    AlignmentTest_FixedSoaElementRef&& ref_b) noexcept  // NOLINT(cppcoreguidelines-rvalue-reference-param-not-moved)
  {
    swap(ref_a, ref_b);
  }

private:
  ::clockwork::FixedSoa<::test::AlignmentTest, size>* soa_;
  size_t index_;
};
