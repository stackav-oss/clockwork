// compare_fields implementation for FixedSoa
template <size_t size>
inline auto ::clockwork::FixedSoa<::test::AlignmentTest, size>::compare_fields(
  const ::clockwork::FixedSoa<::test::AlignmentTest, size>& other, size_t count) const -> bool
{
  if (count == 0)
  {
    return true;
  }
  auto two_byte_this = std::span{::jewels::memory::ObjectPolicy<uint16_t>::ptr(two_byte_[0]), count};
  auto two_byte_other = std::span{::jewels::memory::ObjectPolicy<uint16_t>::ptr(other.two_byte_[0]), count};
  auto eight_byte_this = std::span{::jewels::memory::ObjectPolicy<int64_t>::ptr(eight_byte_[0]), count};
  auto eight_byte_other = std::span{::jewels::memory::ObjectPolicy<int64_t>::ptr(other.eight_byte_[0]), count};
  auto one_byte_this = std::span{::jewels::memory::ObjectPolicy<uint8_t>::ptr(one_byte_[0]), count};
  auto one_byte_other = std::span{::jewels::memory::ObjectPolicy<uint8_t>::ptr(other.one_byte_[0]), count};
  auto four_byte_this = std::span{::jewels::memory::ObjectPolicy<int32_t>::ptr(four_byte_[0]), count};
  auto four_byte_other = std::span{::jewels::memory::ObjectPolicy<int32_t>::ptr(other.four_byte_[0]), count};

  return std::equal(two_byte_this.begin(), two_byte_this.end(), two_byte_other.begin()) &&
         std::equal(eight_byte_this.begin(), eight_byte_this.end(), eight_byte_other.begin()) &&
         std::equal(one_byte_this.begin(), one_byte_this.end(), one_byte_other.begin()) &&
         std::equal(four_byte_this.begin(), four_byte_this.end(), four_byte_other.begin());
}

// wipe_range implementation for FixedSoa
template <size_t size>
void ::clockwork::FixedSoa<::test::AlignmentTest, size>::wipe_range(size_t begin, size_t end)
{
  if (begin >= end)
  {
    return;
  }
  // Zero out the range
  auto two_byte_bytes = std::as_writable_bytes(std::span{two_byte_});
  auto eight_byte_bytes = std::as_writable_bytes(std::span{eight_byte_});
  auto one_byte_bytes = std::as_writable_bytes(std::span{one_byte_});
  auto four_byte_bytes = std::as_writable_bytes(std::span{four_byte_});

  std::fill(
    two_byte_bytes.subspan(begin * sizeof(uint16_t), (end - begin) * sizeof(uint16_t)).begin(),
    two_byte_bytes.subspan(begin * sizeof(uint16_t), (end - begin) * sizeof(uint16_t)).end(),
    std::byte{0});
  std::fill(
    eight_byte_bytes.subspan(begin * sizeof(int64_t), (end - begin) * sizeof(int64_t)).begin(),
    eight_byte_bytes.subspan(begin * sizeof(int64_t), (end - begin) * sizeof(int64_t)).end(),
    std::byte{0});
  std::fill(
    one_byte_bytes.subspan(begin * sizeof(uint8_t), (end - begin) * sizeof(uint8_t)).begin(),
    one_byte_bytes.subspan(begin * sizeof(uint8_t), (end - begin) * sizeof(uint8_t)).end(),
    std::byte{0});
  std::fill(
    four_byte_bytes.subspan(begin * sizeof(int32_t), (end - begin) * sizeof(int32_t)).begin(),
    four_byte_bytes.subspan(begin * sizeof(int32_t), (end - begin) * sizeof(int32_t)).end(),
    std::byte{0});
}

  for (size_t i = begin; i < end; ++i)

    construct_element(i);
  }

// compare_fields implementation for VarSoa
template <size_t max_size>
inline auto ::clockwork::VarSoa<::test::AlignmentTest, max_size>::compare_fields(
  const ::clockwork::VarSoa<::test::AlignmentTest, max_size>& other, size_t count) const -> bool
{
  if (count == 0)
  {
    return true;
  }
  auto two_byte_this = std::span{::jewels::memory::ObjectPolicy<uint16_t>::ptr(two_byte_[0]), count};
  auto two_byte_other = std::span{::jewels::memory::ObjectPolicy<uint16_t>::ptr(other.two_byte_[0]), count};
  auto eight_byte_this = std::span{::jewels::memory::ObjectPolicy<int64_t>::ptr(eight_byte_[0]), count};
  auto eight_byte_other = std::span{::jewels::memory::ObjectPolicy<int64_t>::ptr(other.eight_byte_[0]), count};
  auto one_byte_this = std::span{::jewels::memory::ObjectPolicy<uint8_t>::ptr(one_byte_[0]), count};
  auto one_byte_other = std::span{::jewels::memory::ObjectPolicy<uint8_t>::ptr(other.one_byte_[0]), count};
  auto four_byte_this = std::span{::jewels::memory::ObjectPolicy<int32_t>::ptr(four_byte_[0]), count};
  auto four_byte_other = std::span{::jewels::memory::ObjectPolicy<int32_t>::ptr(other.four_byte_[0]), count};

  return std::equal(two_byte_this.begin(), two_byte_this.end(), two_byte_other.begin()) &&
         std::equal(eight_byte_this.begin(), eight_byte_this.end(), eight_byte_other.begin()) &&
         std::equal(one_byte_this.begin(), one_byte_this.end(), one_byte_other.begin()) &&
         std::equal(four_byte_this.begin(), four_byte_this.end(), four_byte_other.begin());
}

// wipe_range implementation for VarSoa
template <size_t max_size>
void ::clockwork::VarSoa<::test::AlignmentTest, max_size>::wipe_range(size_t begin, size_t end)
{
  if (begin >= end)
  {
    return;
  }
  // Zero out the range
  auto two_byte_bytes = std::as_writable_bytes(std::span{two_byte_});
  auto eight_byte_bytes = std::as_writable_bytes(std::span{eight_byte_});
  auto one_byte_bytes = std::as_writable_bytes(std::span{one_byte_});
  auto four_byte_bytes = std::as_writable_bytes(std::span{four_byte_});

  std::fill(
    two_byte_bytes.subspan(begin * sizeof(uint16_t), (end - begin) * sizeof(uint16_t)).begin(),
    two_byte_bytes.subspan(begin * sizeof(uint16_t), (end - begin) * sizeof(uint16_t)).end(),
    std::byte{0});
  std::fill(
    eight_byte_bytes.subspan(begin * sizeof(int64_t), (end - begin) * sizeof(int64_t)).begin(),
    eight_byte_bytes.subspan(begin * sizeof(int64_t), (end - begin) * sizeof(int64_t)).end(),
    std::byte{0});
  std::fill(
    one_byte_bytes.subspan(begin * sizeof(uint8_t), (end - begin) * sizeof(uint8_t)).begin(),
    one_byte_bytes.subspan(begin * sizeof(uint8_t), (end - begin) * sizeof(uint8_t)).end(),
    std::byte{0});
  std::fill(
    four_byte_bytes.subspan(begin * sizeof(int32_t), (end - begin) * sizeof(int32_t)).begin(),
    four_byte_bytes.subspan(begin * sizeof(int32_t), (end - begin) * sizeof(int32_t)).end(),
    std::byte{0});
}
