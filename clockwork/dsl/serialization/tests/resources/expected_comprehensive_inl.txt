{
  if (count == 0)

    return true;
  }
  auto id_this = std::span{::jewels::memory::ObjectPolicy<uint32_t>::ptr(id_[0]), count};

  auto flag_other = std::span{::jewels::memory::ObjectPolicy<bool>::ptr(other.flag_[0]), count};

  auto measurements_this = std::span{::jewels::memory::ObjectPolicy<::jewels::tap::VarArray<float, 10U>>::ptr(measurements_[0]), count};
  auto measurements_other = std::span{::jewels::memory::ObjectPolicy<::jewels::tap::VarArray<float, 10U>>::ptr(other.measurements_[0]), count};
  auto label_this = std::span{::jewels::memory::ObjectPolicy<::jewels::tap::VarString<32U>>::ptr(label_[0]), count};

  auto optional_score_this = std::span{::jewels::memory::ObjectPolicy<::jewels::tap::Optional<double>>::ptr(optional_score_[0]), count};
  auto optional_score_other = std::span{::jewels::memory::ObjectPolicy<::jewels::tap::Optional<double>>::ptr(other.optional_score_[0]), count};

// wipe_range implementation for FixedSoa
template <size_t size>
void ::clockwork::FixedSoa<::test::ComprehensiveTest, size>::wipe_range(size_t begin, size_t end)
{
  if (begin >= end)
  {
    return;
  }
  // Zero out the range
  auto id_bytes = std::as_writable_bytes(std::span{id_});
  auto timestamp_bytes = std::as_writable_bytes(std::span{timestamp_});
  auto flag_bytes = std::as_writable_bytes(std::span{flag_});
  auto confidence_bytes = std::as_writable_bytes(std::span{confidence_});
  auto position_bytes = std::as_writable_bytes(std::span{position_});
  auto measurements_bytes = std::as_writable_bytes(std::span{measurements_});
  auto label_bytes = std::as_writable_bytes(std::span{label_});
  auto optional_score_bytes = std::as_writable_bytes(std::span{optional_score_});
  auto center_bytes = std::as_writable_bytes(std::span{center_});
